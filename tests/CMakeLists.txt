file(GLOB_RECURSE public_headers RELATIVE "${PROJECT_SOURCE_DIR}/src/SONARENGINE/public" "${PROJECT_SOURCE_DIR}/src/SONARENGINE/public/*.h" "${PROJECT_SOURCE_DIR}/src/SONARENGINE/public/*.hpp")
set(header_sources)
foreach(header IN LISTS public_headers)
  string(MAKE_C_IDENTIFIER "${header}" identifier)
  set(source "${CMAKE_CURRENT_BINARY_DIR}/${identifier}.cpp")
  file(WRITE "${source}" "#include \"${header}\"\n")
  list(APPEND header_sources "${source}")
endforeach()
add_library(public_header_check OBJECT ${header_sources})
target_link_libraries(public_header_check PRIVATE SONARENGINE::SONARENGINE)
add_executable(sdk_contract ../examples/sdk-consumer/main.cpp)
target_link_libraries(sdk_contract PRIVATE SONARENGINE::SONARENGINE)
add_test(NAME sdk_contract COMMAND sdk_contract)
add_executable(graphics_contract graphics_contract.cpp)
target_link_libraries(graphics_contract PRIVATE SONARENGINE::SONARENGINE)
target_compile_definitions(graphics_contract PRIVATE CONTRACT_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
add_test(NAME graphics_contract COMMAND graphics_contract)
add_custom_command(TARGET sdk_contract POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SONARENGINE>" "$<TARGET_FILE_DIR:sdk_contract>")

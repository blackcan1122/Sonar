cmake_minimum_required(VERSION 3.28)
project(Sonar VERSION 0.2.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(FetchContent)
include(CMakePackageConfigHelpers)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(SONAR_BUILD_GAME "Build the SONAR test game" ON)
option(SONAR_RGFW "Use raylib's RGFW desktop backend" OFF)
set(SONAR_RAYLIB_SOURCE "" CACHE PATH "Local raylib source override")
set(SONAR_SPDLOG_SOURCE "" CACHE PATH "Local spdlog source override")
set(SONAR_JSON_SOURCE "" CACHE PATH "Local nlohmann/json source override")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
if(SONAR_RAYLIB_SOURCE)
  add_subdirectory("${SONAR_RAYLIB_SOURCE}" _deps/raylib EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(raylib EXCLUDE_FROM_ALL URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(raylib)
endif()
if(SONAR_RGFW)
  # raylib 5.5 contains RGFW sources but its CMake frontend exposes only GLFW.
  foreach(property IN ITEMS COMPILE_DEFINITIONS INTERFACE_COMPILE_DEFINITIONS)
    get_target_property(backend_definitions raylib ${property})
    list(TRANSFORM backend_definitions REPLACE "^PLATFORM_DESKTOP$" "PLATFORM_DESKTOP_RGFW")
    set_property(TARGET raylib PROPERTY ${property} "${backend_definitions}")
  endforeach()
endif()
if(SONAR_SPDLOG_SOURCE)
  add_subdirectory("${SONAR_SPDLOG_SOURCE}" _deps/spdlog EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(spdlog EXCLUDE_FROM_ALL URL https://github.com/gabime/spdlog/archive/refs/tags/v1.15.3.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(spdlog)
endif()
if(SONAR_JSON_SOURCE)
  add_library(sonar_json INTERFACE)
  target_include_directories(sonar_json INTERFACE "${SONAR_JSON_SOURCE}" "${SONAR_JSON_SOURCE}/include" "${SONAR_JSON_SOURCE}/single_include")
else()
  FetchContent_Declare(json EXCLUDE_FROM_ALL URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(json)
  add_library(sonar_json INTERFACE)
  target_link_libraries(sonar_json INTERFACE nlohmann_json::nlohmann_json)
endif()
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
file(GLOB_RECURSE engine_sources CONFIGURE_DEPENDS src/SONARENGINE/private/*.cpp)
add_library(SONARENGINE SHARED ${engine_sources})
add_library(SONARENGINE::SONARENGINE ALIAS SONARENGINE)
target_compile_features(SONARENGINE PUBLIC cxx_std_20)
target_include_directories(SONARENGINE PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/SONARENGINE/public> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src/SONARENGINE/private)
target_compile_definitions(SONARENGINE PRIVATE SONARENGINE_BUILD $<$<CONFIG:Debug>:DEBUG>)
target_link_libraries(SONARENGINE PRIVATE raylib spdlog::spdlog sonar_json OpenMP::OpenMP_CXX Threads::Threads)
get_target_property(raylib_source raylib SOURCE_DIR)
target_include_directories(SONARENGINE PRIVATE "${raylib_source}")
if(WIN32)
  if(NOT SONAR_RGFW)
    target_compile_definitions(raylib PRIVATE CloseWindow=RaylibCloseWindow ShowCursor=RaylibShowCursor PlaySound=RaylibPlaySound)
    set_source_files_properties(src/SONARENGINE/private/Rendering/Graphics.cpp PROPERTIES COMPILE_DEFINITIONS "CloseWindow=RaylibCloseWindow;ShowCursor=RaylibShowCursor;PlaySound=RaylibPlaySound")
  endif()
  target_compile_definitions(SONARENGINE PRIVATE NOGDI NOUSER WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(SONARENGINE PRIVATE dbghelp)
endif()
if(MSVC)
  target_compile_options(SONARENGINE PRIVATE /utf-8 /bigobj /MP)
  target_compile_definitions(SONARENGINE PRIVATE _CRT_SECURE_NO_WARNINGS)
elseif(WIN32)
  target_compile_options(SONARENGINE PRIVATE -Wa,-mbig-obj)
else()
  target_link_options(SONARENGINE PRIVATE "LINKER:--exclude-libs,ALL")
endif()
if(SONAR_BUILD_GAME)
  file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS src/SONAR/private/*.cpp)
  add_executable(SONAR src/SONAR/main.cpp ${game_sources})
  target_include_directories(SONAR PRIVATE src/SONAR/public)
  target_link_libraries(SONAR PRIVATE SONARENGINE::SONARENGINE OpenMP::OpenMP_CXX)
  target_compile_definitions(SONAR PRIVATE $<$<CONFIG:Debug>:DEBUG>)
  if(MSVC)
    target_compile_options(SONAR PRIVATE /utf-8 /bigobj)
  endif()
  set_target_properties(SONAR PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  add_custom_command(TARGET SONAR POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/resources" "$<TARGET_FILE_DIR:SONAR>/resources" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/src/SONAR/shaders" "$<TARGET_FILE_DIR:SONAR>/src/shaders")
endif()
install(TARGETS SONARENGINE EXPORT SONARENGINETargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/SONARENGINE/public/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SONARENGINETargets NAMESPACE SONARENGINE:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SONARENGINE)
configure_package_config_file(cmake/SONARENGINEConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/SONARENGINEConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SONARENGINE)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SONARENGINEConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SONARENGINEConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/SONARENGINEConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SONARENGINE)
install(DIRECTORY examples/sdk-consumer DESTINATION examples)
install(FILES LICENSE DESTINATION share/SONARENGINE)
install(FILES docs/SDK.md DESTINATION share/SONARENGINE)
get_target_property(spdlog_source spdlog SOURCE_DIR)
get_filename_component(raylib_root "${raylib_source}/.." ABSOLUTE)
install(FILES "${raylib_root}/LICENSE" DESTINATION share/SONARENGINE/licenses RENAME raylib.txt)
install(FILES "${spdlog_source}/LICENSE" DESTINATION share/SONARENGINE/licenses RENAME spdlog.txt)
if(SONAR_JSON_SOURCE)
  set(json_license "${SONAR_JSON_SOURCE}/LICENSE.MIT")
else()
  set(json_license "${json_SOURCE_DIR}/LICENSE.MIT")
endif()
install(FILES "${json_license}" DESTINATION share/SONARENGINE/licenses RENAME json.txt)
if(MINGW)
  get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(runtime IN ITEMS libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll libgomp-1.dll)
    if(EXISTS "${compiler_bin}/${runtime}")
      install(FILES "${compiler_bin}/${runtime}" DESTINATION ${CMAKE_INSTALL_BINDIR})
      file(COPY "${compiler_bin}/${runtime}" DESTINATION "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
    endif()
  endforeach()
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
configure_file(cmake/Package.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/Package.cmake" @ONLY)
set(CPACK_PROJECT_CONFIG_FILE "${CMAKE_CURRENT_BINARY_DIR}/Package.cmake")
set(CPACK_GENERATOR ZIP)
include(CPack)
